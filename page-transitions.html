// Simple page transition wrapper function navigateTo(url) { // Add out class to trigger exit animation document.body.classList.add('transition-fade-out'); // Wait for animation, then navigate setTimeout(() => { window.location.href = url; }, 500); } // On page load, remove out class to trigger enter window.addEventListener('load', () => { document.body.classList.remove('transition-fade-out'); document.body.classList.add('transition-fade-in'); }); // With Barba.js (recommended for complex sites) import barba from '@barba/core'; barba.init({ transitions: [{ name: 'fade', leave(data) { return gsap.to(data.current.container, { opacity: 0, duration: 0.5 }); }, enter(data) { return gsap.from(data.next.container, { opacity: 0, duration: 0.5 }); } }] });
